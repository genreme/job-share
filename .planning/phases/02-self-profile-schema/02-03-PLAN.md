const TargetRoleSchema = z.object({
  id: z.string().uuid(),
  title: z.string().min(1),  // e.g., "Head of Design", "VP Product Design"
  level: z.enum(['ic', 'lead', 'manager', 'director', 'vp', 'c-level']).optional(),

  companyStages: z.array(z.enum(['seed', 'series-a', 'series-b', 'series-c', 'growth', 'public'])).default([]),
  remotePref: z.enum(['remote', 'hybrid', 'onsite', 'flexible']).optional(),
  locations: z.array(z.string()).default([]),  // e.g., ['San Francisco', 'New York']
  salaryRange: z.object({
    min: z.number().optional(),
    max: z.number().optional(),
    currency: z.string().default('USD'),
  }).optional(),
  priorities: z.array(z.string()).default([]),  // e.g., ['impact', 'growth', 'compensation']
  dealbreakers: z.array(z.string()).default([]),  // e.g., ['no-equity', 'travel>50%']

Create mcp-server/src/tools/profile.js following the pattern from jobs.js:

**get_profile tool:**
- Returns the full profile data
- No parameters required
- Returns profile object or error if load fails

**get_experience_by_theme tool:**
- Parameter: theme (string) - tag to filter by
- Returns experience entries where any project has matching tag
- Useful for "show me where you demonstrated leadership"

**get_stories_by_category tool:**
- Parameter: category (string) - questionCategory to filter by
- Returns STAR stories matching the interview question category
- Useful for "find stories about conflict resolution"

**get_skills_by_category tool:**
- Parameter: category (string) - skill category to filter
- Returns skills in that category with their evidence references
- Useful for "what technical skills do you have?"

**Structure following jobs.js pattern:**
```javascript
export async function get_profile() {
  const profile = loadProfile();
  return profile;
}

export async function get_experience_by_theme({ theme }) {
  const profile = loadProfile();
  return profile.experience.filter(exp =>
    exp.projects.some(proj => proj.tags.includes(theme))
  );
}

// etc.
```
